// --- ERROR BOUNDARY ---
class ErrorBoundary extends Component<{ children: React.ReactNode }, { hasError: boolean, error: Error | null }> {
  constructor(props: { children: React.ReactNode }) {
    super(props);
    this.state = { hasError: false, error: null };
  }
  static getDerivedStateFromError(error: Error) { return { hasError: true, error }; }
  componentDidCatch(error: Error, errorInfo: ErrorInfo) { console.error("AICE PROTOCOL EXCEPTION:", error, errorInfo); }
  render() {
    if (this.state.hasError) {
      return (
        <div className="min-h-screen bg-black text-red-500 p-10 font-mono flex flex-col items-center justify-center text-center z-[9999] relative">
          <div className="border border-red-500/50 p-12 rounded-lg bg-red-950/10 backdrop-blur-xl max-w-2xl w-full">
            <AlertTriangle size={64} className="mx-auto mb-6 text-red-500 animate-pulse" />
            <h1 className="text-4xl mb-4 font-black tracking-widest">SYSTEM CRITICAL ERROR</h1>
            <p className="mb-8 text-white/80 font-sans">The protocol encountered a critical deterministic exception.</p>
            <button onClick={() => window.location.reload()} className="px-8 py-4 bg-red-600 text-white rounded hover:bg-red-500 transition-all font-bold uppercase tracking-[0.2em]">Reboot System</button>
          </div>
        </div>
      );
    }
    return this.props.children;
  }
}

const LoginModal: React.FC<{ isOpen: boolean, onClose: () => void, user: FirebaseUser | null, initialMode?: 'register' | 'login' }> = ({ isOpen, onClose, user, initialMode = 'login' }) => {
  const [isRegistering, setIsRegistering] = useState(initialMode === 'register');
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [showPassword, setShowPassword] = useState(false); 
  const [error, setError] = useState("");
  const [successMsg, setSuccessMsg] = useState("");
  const [loading, setLoading] = useState(false);

  useEffect(() => { 
      setIsRegistering(initialMode === 'register'); 
      setError(""); 
      setSuccessMsg("");
  }, [initialMode, isOpen]);

  const logUserAccess = async (user: FirebaseUser) => {
    try {
      const userRef = doc(db, 'artifacts', appId, 'users', user.uid, 'profile', 'data');
      await setDoc(userRef, { email: user.email, lastLogin: serverTimestamp(), uid: user.uid, displayName: user.displayName || 'Operative', role: 'user' }, { merge: true });
    } catch (e) { console.error(e); }
  };

  const mapAuthError = (errCode: string) => {
      if (errCode.includes('password')) return "ACCESS DENIED: INVALID CREDENTIALS";
      if (errCode.includes('email')) return "ERROR: INVALID EMAIL FORMAT";
      if (errCode.includes('not-found')) return "USER NOT FOUND";
      if (errCode.includes('in-use')) return "ID ALREADY REGISTERED";
      return "SYSTEM ERROR: TRY AGAIN";
  };

  const handleGoogleLogin = async () => {
    setLoading(true); setError("");
    try {
      const result = await signInWithPopup(auth, googleProvider);
      await logUserAccess(result.user);
      onClose();
    } catch (err: any) { setError(mapAuthError(err.message)); } finally { setLoading(false); }
  };

  const handleEmailAuth = async () => {
    setLoading(true); setError(""); setSuccessMsg("");
    
    if (isRegistering && password !== confirmPassword) {
        setError("SECURITY ALERT: PASSWORDS DO NOT MATCH");
        setLoading(false);
        return;
    }
    
    if (password.length < 6) {
        setError("SECURITY ALERT: PASSWORD TOO WEAK (MIN 6 CHARS)");
        setLoading(false);
        return;
    }

    try {
      let result;
      if (isRegistering) result = await createUserWithEmailAndPassword(auth, email, password);
      else result = await signInWithEmailAndPassword(auth, email, password);
      await logUserAccess(result.user);
      onClose();
    } catch (err: any) { setError(mapAuthError(err.message)); } finally { setLoading(false); }
  };

  const handleResetPassword = async () => {
      if (!email) { setError("ENTER ID (EMAIL) TO RESET"); return; }
      try {
          await sendPasswordResetEmail(auth, email);
          setSuccessMsg("RESET LINK TRANSMITTED TO INBOX");
          setError("");
      } catch (err: any) { setError(mapAuthError(err.message)); }
  };

  if (!isOpen) return null;
  if (user) return null;

  return (
    <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/95 backdrop-blur-xl p-4">
      <div className="w-full max-w-md bg-[#050505] border border-[#00F3FF] shadow-[0_0_100px_rgba(0,243,255,0.15)] rounded-2xl relative overflow-hidden p-8 animate-in zoom-in duration-300">
        <button onClick={onClose} className="absolute top-4 right-4 text-white/30 hover:text-[#00F3FF]"><X /></button>
        
        <div className="text-center mb-8">
            <div className="inline-flex justify-center items-center w-16 h-16 rounded-full bg-[#00F3FF]/10 border border-[#00F3FF] mb-4 shadow-[0_0_20px_rgba(0,243,255,0.3)]">
                <ShieldCheck size={32} className="text-[#00F3FF]" />
            </div>
            <h2 className="text-2xl font-bold text-white uppercase tracking-widest mb-1">{isRegistering ? 'New Operative' : 'Identity Verification'}</h2>
            <p className="text-xs text-[#00F3FF] font-mono tracking-wider">A.I.C.E. SECURITY PROTOCOL v2.0</p>
        </div>

        {error && <div className="bg-red-500/10 border border-red-500/50 p-3 mb-6 text-red-500 text-xs font-mono flex items-center gap-2 animate-pulse"><AlertTriangle size={12}/> {error}</div>}
        {successMsg && <div className="bg-green-500/10 border border-green-500/50 p-3 mb-6 text-green-500 text-xs font-mono flex items-center gap-2"><CheckCircle2 size={12}/> {successMsg}</div>}

        <div className="space-y-4">
            <button onClick={handleGoogleLogin} disabled={loading} className="w-full py-3 bg-white text-black font-bold uppercase tracking-wider rounded flex items-center justify-center gap-3 hover:bg-gray-200 transition-colors">
                <img src="https://www.gstatic.com/firebasejs/ui/2.0.0/images/auth/google.svg" className="w-5 h-5" alt="G" />
                {loading ? 'Scanning...' : 'Sign in with Google'}
            </button>
            <div className="flex items-center gap-4 my-4"><div className="h-[1px] bg-white/10 flex-1"></div><span className="text-white/30 text-[10px] uppercase">OR ENCRYPTED KEY</span><div className="h-[1px] bg-white/10 flex-1"></div></div>

            <div className="space-y-3">
                <div className="relative">
                    <Mail className="absolute left-3 top-3 text-white/30" size={18} />
                    <input type="email" placeholder="OPERATIVE ID (EMAIL)" value={email} onChange={(e) => setEmail(e.target.value)} className="w-full bg-black border border-white/20 rounded p-3 pl-10 text-white text-sm outline-none focus:border-[#00F3FF] font-mono transition-colors" />
                </div>
                
                {/* PASSWORD FIELD WITH TOGGLE */}
                <div className="relative">
                    <LockKeyhole className="absolute left-3 top-3 text-white/30" size={18} />
                    <input 
                        type={showPassword ? "text" : "password"} 
                        placeholder="ACCESS CODE (PASSWORD)" 
                        value={password} 
                        onChange={(e) => setPassword(e.target.value)} 
                        className="w-full bg-black border border-white/20 rounded p-3 pl-10 pr-10 text-white text-sm outline-none focus:border-[#00F3FF] font-mono transition-colors" 
                    />
                    <button onClick={() => setShowPassword(!showPassword)} className="absolute right-3 top-3 text-white/30 hover:text-white">
                        {showPassword ? <Eye size={18} /> : <Eye className="opacity-50" size={18} />}
                    </button>
                </div>
                
                {isRegistering && (
                    <div className="relative animate-in slide-in-from-top-2 fade-in">
                        <LockKeyhole className="absolute left-3 top-3 text-[#00F3FF]" size={18} />
                        <input 
                            type={showPassword ? "text" : "password"} 
                            placeholder="CONFIRM ACCESS CODE" 
                            value={confirmPassword} 
                            onChange={(e) => setConfirmPassword(e.target.value)} 
                            className="w-full bg-black border border-[#00F3FF]/50 rounded p-3 pl-10 text-white text-sm outline-none focus:border-[#00F3FF] font-mono shadow-[0_0_10px_rgba(0,243,255,0.1)]" 
                        />
                    </div>
                )}
            </div>

            {!isRegistering && (
                <div className="flex justify-end">
                    <button onClick={handleResetPassword} className="text-[10px] text-white/50 hover:text-[#00F3FF] uppercase tracking-wider transition-colors">
                        Forgot Access Code?
                    </button>
                </div>
            )}

            <button onClick={handleEmailAuth} disabled={loading} className="w-full py-4 bg-[#00F3FF]/10 border border-[#00F3FF] text-[#00F3FF] font-black uppercase tracking-[0.2em] rounded hover:bg-[#00F3FF] hover:text-black transition-all shadow-[0_0_20px_rgba(0,243,255,0.1)] mt-2">
                {loading ? 'AUTHENTICATING...' : (isRegistering ? 'INITIALIZE ACCOUNT' : 'VERIFY IDENTITY')}
            </button>
        </div>

        <div className="mt-6 text-center">
            <button onClick={() => { setIsRegistering(!isRegistering); setError(""); setSuccessMsg(""); }} className="text-white/40 text-xs hover:text-[#00F3FF] underline decoration-dotted underline-offset-4">
                {isRegistering ? 'Return to Login Interface' : 'Request New Clearance Level'}
            </button>
        </div>
      </div>
    </div>
  );
};

                      <div className="bg-white/5 px-4 py-2 border-b border-white/10 text-[8px] text-gray-500 font-mono uppercase tracking-widest">Level 5 Access</div>
                      <button onClick={(e) => { e.stopPropagation(); setView('INSTITUTIONAL'); }} className="w-full text-left px-4 py-3 text-[10px] font-mono font-bold tracking-widest text-white hover:bg-white hover:text-black transition-all flex items-center gap-3"><ShieldCheck size={12} className="text-[#00F3FF]" /> INSTITUTIONAL</button>
                    </div>
                  </div>
                )}
              </div>
            ))}
          </div>
          <div className="w-[1px] h-6 bg-white/20"></div>
          <div className="flex items-center gap-4">
              <button onClick={() => setView('SALES')} className="px-6 py-2 bg-[#00F3FF] text-black rounded-full text-[10px] font-black tracking-widest uppercase hover:bg-white hover:scale-105 transition-all shadow-[0_0_20px_rgba(0,243,255,0.3)]">ACQUIRE 🐦‍🔥</button>
              {!user && <button onClick={() => setIsLoginOpen(true)} className="px-6 py-2 border border-white/20 rounded-full text-[10px] font-black tracking-widest uppercase hover:bg-white hover:text-black transition-all">LOGIN</button>}
              {user && <button onClick={() => setView('PROFILE')} className="w-8 h-8 rounded-full bg-green-500/20 border border-green-500 flex items-center justify-center text-green-500 hover:bg-green-500 hover:text-black transition-all"><UserCheck size={14}/></button>}
          </div>
        </div>
      </nav>

      {/* --- CONTENT LAYOUT --- */}
      {/* flex-grow pushes footer down. min-h-full ensures it fills the scroll container */}
      <div className="flex flex-col min-h-full relative z-10">
          <main className="w-full flex-grow flex flex-col pt-24">
            {view === 'HOME' && <HomeView setView={setView} />}
            {view === 'ADAPTIVE' && <AdaptiveView />}
            {view === 'INTELLIGENCE' && <IntelligenceView />}
            {view === 'ENTROPY' && <EntropyView />}
            {view === 'CONTROL' && <ControlView />}
            {view === 'SYSTEM' && <SystemView />}
            
            {view === 'SALES' && (
                <AccessTerminal 
                    setView={setView} 
                    onBuy={handlePayment} 
                    onOpenDoc={setActiveDoc} 
                    user={user} 
                    openLogin={() => { setLoginMode('login'); setIsLoginOpen(true); }} 
                />
            )}
            {view === 'BOOK' && <BookView onBuy={handlePayment} />}
            {view === 'PROFILE' && <ProfileView user={user} />}
            {view === 'ADMIN' && <AdminView />}
            {view === 'HOW_IT_WORKS' && <HowItWorksView />}
            {view === 'DEVELOPERS' && <DeveloperView />}
            {view === 'PILOT_PROGRAMS' && <PilotProgramView onOpenDoc={setActiveDoc} setView={setView} />}
            {view === 'PHOENIX_BOT' && <PhoenixBotView />}
            {view === 'INSTITUTIONAL' && <InstitutionalView />}
            {view === 'PILOT_PHOENIX' && <Pilot_Program onBuy={handlePayment} />}
          </main>

          <Footer 
            setView={setView} 
            onOpenDoc={setActiveDoc} 
            openLogin={(mode) => { setLoginMode(mode); setIsLoginOpen(true); }} 
            user={user} 
          />
      </div>
      
      {/* MODALS */}
      {/* NEW LINE - THE PDF BRIDGE */}
      <DocumentModal 
        isOpen={activeDoc !== null && docData.fileUrl !== ''} 
        onClose={() => setActiveDoc(null)} 
        title={docData.title} 
        fileUrl={docData.fileUrl} 
      />
      <LoginModal isOpen={isLoginOpen} onClose={() => setIsLoginOpen(false)} user={user} initialMode={loginMode} />
      
      {/* --- CHANGE THIS LINE BELOW --- */}
      <GlobalChat /> 
      {/* ----------------------------- */}
      
      <div className="fixed bottom-8 left-8 z-[90]">
         <button onClick={() => setActiveDoc('WHITEPAPER')} className="flex items-center gap-2 px-4 py-2 bg-black/80 border border-white/10 rounded-lg text-white/50 text-xs font-mono uppercase tracking-widest hover:text-[#00F3FF] hover:border-[#00F3FF] transition-all shadow-lg backdrop-blur-sm">
            <BookOpen size={14} /> WHITE PAPER
         </button>
      </div>
    </div>
  );
};

const App: React.FC = () => <ErrorBoundary><AppContent /></ErrorBoundary>;

export default App;
