// --- The Neural Earth Component ---
const NeuralEarth: React.FC<{ visible: boolean }> = ({ visible }) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let width = window.innerWidth;
    let height = window.innerHeight;
    canvas.width = width;
    canvas.height = height;

    const particles: {x: number, y: number, vx: number, vy: number, size: number}[] = [];
    const particleCount = 50; 
    
    for (let i = 0; i < particleCount; i++) {
      particles.push({
        x: Math.random() * width,
        y: Math.random() * height,
        vx: (Math.random() - 0.5) * 0.15,
        vy: (Math.random() - 0.5) * 0.15,
        size: Math.random() * 2.5
      });
    }

    const animate = () => {
      ctx.clearRect(0, 0, width, height);
      particles.forEach((p, i) => {
        p.x += p.vx;
        p.y += p.vy;

        if (p.x < 0) p.x = width;
        if (p.x > width) p.x = 0;
        if (p.y < 0) p.y = height;
        if (p.y > height) p.y = 0;

        ctx.fillStyle = '#00F3FF';
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();

        for (let j = i; j < particles.length; j++) {
          const p2 = particles[j];
          const dx = p.x - p2.x;
          const dy = p.y - p2.y;
          const dist = Math.sqrt(dx*dx + dy*dy);
          if (dist < 150) {
            ctx.strokeStyle = `rgba(0, 243, 255, ${0.15 - dist/1000})`;
            ctx.lineWidth = 0.8;
            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(p2.x, p2.y);
            ctx.stroke();
          }
        }
      });
      requestAnimationFrame(animate);
    };
    animate();

    const handleResize = () => {
      width = window.innerWidth;
      height = window.innerHeight;
      canvas.width = width;
      canvas.height = height;
    };
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);

  return (
    <div className={`fixed inset-0 z-0 overflow-hidden pointer-events-none transition-opacity duration-1000 ${visible ? 'opacity-100' : 'opacity-0'}`}>
      <canvas ref={canvasRef} className="absolute inset-0 z-10" />
    </div>
  );
};

             </h1>
             <p className="mt-10 text-white/90 font-['Space_Mono'] text-xl md:text-2xl tracking-widest uppercase opacity-80 font-bold">
               Thermodynamic regularization for the singularity.
             </p>
          </div>

          {/* WIDER GRID to reveal the background center */}

            {/* LEFT SIDE: ORANGE / INSTABILITY */}

                <button 
                  onClick={() => setView('SALES')}
                  className="group relative px-12 py-6 bg-[#FFD700] text-black font-['Space_Mono'] font-bold tracking-widest hover:bg-white transition-all skew-x-[-10deg] shadow-[0_0_40px_rgba(255,215,0,0.4)] text-xl"
                >
                  <div className="skew-x-[10deg] flex items-center gap-4">
                    INITIATE PROTOCOL <ChevronRight size={24} />
                  </div>
                  <div className="absolute inset-0 bg-white/50 animate-pulse opacity-0 group-hover:opacity-100 transition-opacity"></div>
                </button>
                <div className="mt-16 flex items-center gap-4 text-white/60 text-sm font-['Space_Mono'] animate-bounce font-bold tracking-widest">

               <h3 className="text-[#00F3FF] font-['Rajdhani'] font-bold text-4xl uppercase tracking-widest flex items-center gap-4 justify-end">
                 The Regulation <ShieldCheck size={36} />
               </h3>
               <p className="font-['Inter'] text-2xl text-gray-100 leading-relaxed font-light">
                 The AIEC Protocol acts as a Thermodynamic Safety Governor. We enforce a physics-based "Veto" on information processing, making catastrophic failure energetically impossible.
               </p>
            </div>

          </div>
        </div>
      </div>

      {/* SECTION 1: THE VARIABLE (Entropy Definition)
        ANIMATION: Slide from RIGHT (Red Theme)
      */}

         <div className="max-w-[90rem] mx-auto w-full grid grid-cols-1 md:grid-cols-2 gap-32 items-center">
            {/* Left Reveal Box */}
            <DiagonalRevealBox progress={reveal1} color="#FF4500">
               <h3 className="text-[#FF4500] font-['Space_Mono'] tracking-widest text-lg mb-10 font-bold">THE VARIABLE</h3>
               <div className="text-7xl md:text-9xl font-['Rajdhani'] font-bold text-white mb-10 leading-none">
                 SYSTEMIC<br/>ENTROPY
               </div>
               <p className="font-['Inter'] text-gray-100 leading-loose text-2xl mb-8 font-light">
                 Entropy isn't just a physics concept; it is the silent killer of efficiency. It is the mathematical measure of how close your system is to a chaotic breakdown.
               </p>
            </DiagonalRevealBox>

            {/* Right Animated Card (Red) */}
            <div className="flex justify-center md:justify-end" style={card1Style}>
               <div className="relative w-full max-w-2xl p-3 bg-gradient-to-br from-[#FF4500]/40 to-transparent rounded-3xl shadow-[0_0_100px_rgba(255,69,0,0.25)]">
                 <div className="bg-[#050505]/80 backdrop-blur-xl p-16 rounded-2xl border border-[#FF4500]/30 flex flex-col items-center text-center h-full relative overflow-hidden">
                   <div className="absolute top-0 inset-x-0 h-2 bg-gradient-to-r from-transparent via-[#FF4500] to-transparent opacity-50"></div>
                   
                   <div className="mb-10 p-8 rounded-full bg-[#FF4500]/10 border border-[#FF4500]/20 shadow-[0_0_40px_rgba(255,69,0,0.3)]">
                      <AlertTriangle size={64} className="text-[#FF4500]" />
                   </div>

                   <h2 className="text-7xl font-['Rajdhani'] font-bold text-white mb-10 drop-shadow-lg leading-tight">DISORDER IS<br/>DEFAULT.</h2>
                   <p className="font-['Space_Mono'] text-gray-200 text-lg leading-relaxed mb-12">
                      Without external work, every system drifts towards chaos. This is physics, not an opinion.
                   </p>
                   <div className="mt-6 flex items-center gap-4 text-[#FF4500] text-sm font-['Space_Mono'] tracking-widest font-bold">
                     <Activity size={20} />
                     ENTROPY RISING
                   </div>
                 </div>
               </div>
            </div>
         </div>
      </div>

      {/* SECTION 2: THE LAW (Prevention)
        ANIMATION: Slide from LEFT (Dark Blue Theme)
      */}
      <div className="min-h-[140vh] flex items-center justify-center px-6 py-24 relative z-20">
         <div className="max-w-[90rem] mx-auto w-full grid grid-cols-1 md:grid-cols-2 gap-32 items-center">
            
            {/* Left Animated Card (Blue) */}
            <div className="flex justify-center md:justify-start order-2 md:order-1" style={card2Style}>
               <div className="relative w-full max-w-2xl p-3 bg-gradient-to-br from-[#3B82F6]/40 to-transparent rounded-3xl shadow-[0_0_100px_rgba(59,130,246,0.25)]">
                 <div className="bg-[#050505]/80 backdrop-blur-xl p-16 rounded-2xl border border-[#3B82F6]/30 flex flex-col items-center text-center h-full relative overflow-hidden">
                   <div className="absolute top-0 inset-x-0 h-2 bg-gradient-to-r from-transparent via-[#3B82F6] to-transparent opacity-50"></div>
                   
                   <div className="mb-10 p-8 rounded-full bg-[#3B82F6]/10 border border-[#3B82F6]/20 shadow-[0_0_40px_rgba(59,130,246,0.3)]">
                      <Shield size={64} className="text-[#3B82F6]" />
                   </div>

                   <h2 className="text-7xl font-['Rajdhani'] font-bold text-white mb-10 drop-shadow-lg leading-tight">PREVENTING<br/>FAILURE.</h2>
                   <p className="font-['Space_Mono'] text-gray-200 text-lg leading-relaxed mb-12">
                      The glass does not have to shatter. We reinforce the structure before stress accumulation occurs.
                   </p>
                   <div className="mt-6 flex items-center gap-4 text-[#3B82F6] text-sm font-['Space_Mono'] tracking-widest font-bold">
                     <Lock size={20} />
                     STRUCTURAL INTEGRITY 100%
                   </div>
                 </div>
               </div>
            </div>

            {/* Right Reveal Box (Blue) */}
            <DiagonalRevealBox progress={reveal2} color="#3B82F6" className="order-1 md:order-2">
               <h3 className="text-[#3B82F6] font-['Space_Mono'] tracking-widest text-lg mb-10 font-bold">THE LAW</h3>
               <div className="text-7xl md:text-9xl font-['Rajdhani'] font-bold text-white mb-10 leading-none">
                 MATHEMATICAL<br/>CERTAINTY
               </div>
               <p className="font-['Inter'] text-gray-100 leading-loose text-2xl mb-8 font-light">
                 <strong>Second Law of Thermodynamics:</strong> In any closed system—be it a power grid or a trading floor—disorder naturally increases.
               </p>
            </DiagonalRevealBox>
         </div>
      </div>

      {/* SECTION 3: THE APPLICATION (Solution)
        ANIMATION: Slide from RIGHT (Cyan Theme)
      */}
      <div className="min-h-[140vh] flex items-center justify-center px-6 py-24 relative z-20 overflow-hidden">
         <div className="max-w-[90rem] mx-auto w-full grid grid-cols-1 md:grid-cols-2 gap-32 items-center">
            
            {/* Left Reveal Box (Cyan) */}
            <DiagonalRevealBox progress={reveal3} color="#00F3FF">
               <h3 className="text-[#00F3FF] font-['Space_Mono'] tracking-widest text-lg mb-10 font-bold">THE CONTEXT</h3>
               <div className="text-7xl md:text-9xl font-['Rajdhani'] font-bold text-white mb-10 leading-none">
                 COGNITIVE<br/>VELOCITY
               </div>
               <p className="font-['Inter'] text-gray-100 leading-loose text-2xl mb-8 font-light">
                 When information flow exceeds cognitive processing limits, you hit the "Hallucination Horizon".
               </p>
            </DiagonalRevealBox>

            {/* Right Animated Card (Cyan) */}
            <div className="flex justify-center md:justify-end" style={card3Style}>
               <div className="relative w-full max-w-2xl p-3 bg-gradient-to-br from-[#00F3FF]/40 to-transparent rounded-3xl shadow-[0_0_100px_rgba(0,243,255,0.2)]">

                   <div className="absolute top-0 inset-x-0 h-2 bg-gradient-to-r from-transparent via-[#00F3FF] to-transparent opacity-50"></div>
                   
                   <div className="mb-10 p-8 rounded-full bg-[#00F3FF]/10 border border-[#00F3FF]/20 shadow-[0_0_40px_rgba(0,243,255,0.3)]">
                      <ShieldCheck size={64} className="text-[#00F3FF]" />
                   </div>

                   <h2 className="text-7xl font-['Rajdhani'] font-bold text-white mb-10 drop-shadow-lg leading-tight">THE AIEC<br/>SOLUTION.</h2>
                   <p className="font-['Space_Mono'] text-gray-200 text-lg leading-relaxed mb-12">
                      We enforce a Thermodynamic Veto on information processing. We cap the risk before it spreads.
                   </p>
                   
                   <button 
                     onClick={() => setView('SALES')}
                     className="w-full py-6 bg-white text-black font-['Space_Mono'] font-bold tracking-[0.2em] hover:bg-[#FFD700] transition-colors uppercase text-base shadow-lg"
                   >
                     ACQUIRE THE PROTOCOL
                   </button>

                 </div>
               </div>
            </div>

         </div>
      </div>

    </div>
  );
};
